% =============================================================================
% Warning environment
% =============================================================================

\usepackage{pifont}
\usepackage{mdframed}

\newenvironment{warning}
{%
	\samepage
	\par
	\begin{mdframed}[
		linewidth=2pt,
		linecolor=mdColorAlert
		]
		\begin{list}{}{%
				\leftmargin=1cm
				\labelwidth=\leftmargin
			}%
			\item[\color{mdColorAlert}\Large\ding{54}]%
		}
		{%
		\end{list}
	\end{mdframed}
	\par
}

\newenvironment{note}
{%
	\samepage
	\par
	\begin{mdframed}[
		linewidth=2pt,
		linecolor=mdColorPrimary
		]
		\begin{list}{}{%
				\leftmargin=1cm
				\labelwidth=\leftmargin
			}%
			\item[\Large\ding{43}]%
		}
		{%
		\end{list}
	\end{mdframed}
	\par
}


% =============================================================================
% Verbatim environments
%
% These environments are used for source-code and documentation examples.
% Machine-generated output should normally use mdlayout's Printout
% environment instead.
% =============================================================================

\usepackage{fvextra}


% -----------------------------------------------------------------------------
% Gray verbatim, normal text width
% -----------------------------------------------------------------------------

\DefineVerbatimEnvironment{grayverbatim}{Verbatim}{
	fontsize=\small,
	frame=none,
	rulecolor=\color{gray},
	framesep=0mm,
	baselinestretch=1,
	bgcolor=gray!10
}


% -----------------------------------------------------------------------------
% Gray verbatim, extended mdlayout width
% -----------------------------------------------------------------------------

\DefineVerbatimEnvironment{graywideverbatim}{Verbatim}{
	fontsize=\small,
	frame=none,
	rulecolor=\color{gray},
	framesep=0mm,
	baselinestretch=1,
	bgcolor=gray!10,
	xleftmargin=\megatextindent,
	xrightmargin=0pt
}


% -----------------------------------------------------------------------------
% Source-code listing
%
% This is intentionally distinct from Printout:
% listing = author-controlled source code
% Printout = machine-generated output
% -----------------------------------------------------------------------------

\DefineVerbatimEnvironment{listing}{Verbatim}{
	formatcom=\color{black}\footnotesize,
	numbers=left,
	numbersep=4pt,
	frame=single,
	rulecolor=\color{mdColorPrimary},
	xleftmargin=\megatextindent,
	xrightmargin=0pt
}


% -----------------------------------------------------------------------------
% Generic wide verbatim
% -----------------------------------------------------------------------------

\DefineVerbatimEnvironment{wideverbatim}{Verbatim}{
	xleftmargin=\megatextindent
}
